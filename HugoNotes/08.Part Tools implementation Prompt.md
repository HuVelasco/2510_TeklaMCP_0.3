  Beam Creation Template:
  def create_beam_code(start, end, profile, material, position_params):
      return f"""
  var beam = new Beam();
  beam.StartPoint = new Tekla.Structures.Geometry3d.Point({start});
  beam.EndPoint = new Tekla.Structures.Geometry3d.Point({end});
  beam.Profile.ProfileString = "{profile}";
  beam.Material.MaterialString = "{material}";
  {generate_position_code(**position_params)}
  beam.Insert();
  model.CommitChanges();
  return "Beam created - ID: " + beam.Identifier.ID.ToString();
  """

  Column Creation (uses Beam class with vertical geometry):
  def create_column_code(base_point, height, profile, material, position_params):
      x, y, z = base_point.split()
      return f"""
  var beam = new Beam();
  beam.StartPoint = new Tekla.Structures.Geometry3d.Point({x}, {y}, {z});
  beam.EndPoint = new Tekla.Structures.Geometry3d.Point({x}, {y}, {z} + {height});
  beam.Profile.ProfileString = "{profile}";
  beam.Material.MaterialString = "{material}";
  {generate_position_code(**position_params)}
  beam.Insert();
  model.CommitChanges();
  return "Column created - ID: " + beam.Identifier.ID.ToString();
  """

  ContourPlate Creation:
  def create_contour_plate_code(points, thickness, material):
      point_lines = []
      for point in points.split(';'):
          x, y, z = point.split(',')
          point_lines.append(f"plate.AddContourPoint(new Tekla.Structures.Geometry3d.Point({x}, {y}, {z}));")

      return f"""
  var plate = new ContourPlate();
  {chr(10).join(point_lines)}
  plate.Profile.ProfileString = "PL{thickness}";
  plate.Material.MaterialString = "{material}";
  plate.Insert();
  model.CommitChanges();
  return "Plate created - ID: " + plate.Identifier.ID.ToString();
  """
