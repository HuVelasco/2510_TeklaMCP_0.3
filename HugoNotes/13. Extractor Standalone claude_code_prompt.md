# TEKLA DRAWING EXTRACTION TOOL - REFACTOR & STANDALONE APPLICATION

## PROJECT OVERVIEW

You are refactoring the Tekla drawing extraction functionality from an MCP server into a **standalone Windows GUI application** for ML training data generation. The extracted data will train ML models to predict dimension placement on shop drawings.

## CONTEXT & REQUIREMENTS

### Current State
- Extraction code is embedded in `ProgramClean.cs` MCP server (lines 1106-2800+)
- Outputs JSONL format with good geometry data but missing critical properties
- Hard to visualize and navigate the JSON output
- Contains debug noise and unparsed placeholder values

### Your Mission
Create a **standalone WinForms/WPF application** that:
1. Extracts Tekla shop drawing data for ML training
2. Fixes all data quality issues identified in the audit
3. Organizes JSON output for easy human visualization
4. Provides user-friendly parameter controls
5. Removes extraction code from MCP (keep MCP simple)

#### 1. Extract Dimension Values
**Current Issue:** 
```json
"Value": "Tekla.Structures.Drawing.ContainerElement"
```

**Fix:**
```csharp
// For StraightDimension
var dimensionValue = dimension.Value;
if (dimensionValue != null)
{
    // Get the full text content from ContainerElement
    var containerElement = dimensionValue as Tekla.Structures.Drawing.ContainerElement;
    if (containerElement != null)
    {
        // This gives us the complete dimension text
        dimensionData["value_text"] = containerElement.GetUnformattedString();
    }
    
    // Get prefix and postfix from attributes
    var attributes = dimension.Attributes as Tekla.Structures.Drawing.StraightDimension.StraightDimensionAttributes;
    if (attributes != null)
    {
        dimensionData["prefix"] = attributes.DimensionValuePrefix ?? "";
        dimensionData["postfix"] = attributes.DimensionValuePostfix ?? "";
    }
    
    // Also calculate from coordinates as backup
    var start = ParsePoint(dimension.StartPoint);
    var end = ParsePoint(dimension.EndPoint);
    double calculatedValue = Math.Sqrt(
        Math.Pow(end.X - start.X, 2) + 
        Math.Pow(end.Y - start.Y, 2)
    );
    dimensionData["value_calculated"] = Math.Round(calculatedValue, 2);
}

// For other dimensions: Similar fix...
```

#### 2. Extract Part Profile, Material, Length
**Current Issue:** `"model_object": {}` is empty

**Fix:**
```csharp
// Cache: GUID → Static Attributes
Dictionary<Guid, Hashtable> staticCache = new Dictionary<Guid, Hashtable>();

foreach (var part in drawingParts)
{
    // Get GUID from ModelObject (correct way)
    Guid guid = part.Identifier.GUID;
    
    // Only extract if not already cached
    if (!staticCache.ContainsKey(guid))
    {
        Hashtable staticAttrs = new Hashtable();
        
        // Static attributes (extract once)
        ArrayList strings = new ArrayList() { 
            "NAME", "MATERIAL", "FINISH", 
            "ASSEMBLY_POSITION_CODE", "PROFILE", 
            "PART_POS", "ASSEMBLY_POS", "PROFILE_TYPE" 
        };
        
        ArrayList doubles = new ArrayList() {
            "START_X", "START_Y", "START_Z",  // Global coordinates
            "END_X", "END_Y", "END_Z",        // Global coordinates
            "TOP_LEVEL_GLOBAL", "BOTTOM_LEVEL_GLOBAL"
        };
        
        part.GetAllReportProperties(strings, doubles, new ArrayList(), ref staticAttrs);
        
        // Add GUID to the attributes (optional)
        staticAttrs["GUID"] = guid.ToString();
        
        // Store in cache
        staticCache[guid] = staticAttrs;
    }
    // else: Part already in cache, skip extraction
}

// Later, when processing views:
foreach (var view in views)
{
    // Set work plane...
    
    foreach (var part in drawingParts)
    {
        Guid guid = part.Identifier.GUID;
        Hashtable staticData = staticCache[guid];  // Get from cache

        // Save current work plane
        TransformationPlane originalPlane = model.GetWorkPlaneHandler().GetCurrentTransformationPlane();
        
        // Set work plane to view coordinates (THIS IS THE KEY)
        model.GetWorkPlaneHandler().SetCurrentTransformationPlane(view.ViewCoordinateSystem);
        
        // Get LOCAL attributes (view coordinates)
        Hashtable localAttrs = new Hashtable();
        ArrayList localDoubles = new ArrayList() {
            "START_X_IN_WORK_PLANE", "START_Y_IN_WORK_PLANE", "START_Z_IN_WORK_PLANE",
            "END_X_IN_WORK_PLANE", "END_Y_IN_WORK_PLANE", "END_Z_IN_WORK_PLANE",
            "TOP_LEVEL", "BOTTOM_LEVEL"
        };
        
        part.GetAllReportProperties(new ArrayList(), localDoubles, new ArrayList(), ref localAttrs);
        
        // Now you have:
        // staticData contains global attributes (extracted once)
        // localAttrs contains view-specific coordinates (extracted per view)

        // Restore original work plane
        model.GetWorkPlaneHandler().SetCurrentTransformationPlane(originalPlane);
    }
}



```

#### 3. Extract Bolt Positions
**Fix:**
```csharp
if (obj is Tekla.Structures.Drawing.Bolt drawingBolt)
{
    var boltModelObj = drawingBolt.ModelObject;
    if (boltModelObj != null)
    {
        var boltGroup = new Tekla.Structures.Model.Model()
            .SelectModelObject(boltModelObj) as Tekla.Structures.Model.BoltGroup;
        
        if (boltGroup != null)
        {
            var positions = new List<object>();
            foreach (Point pos in boltGroup.BoltPositions) 
// Gets the single bolt positions. All the bolt positions are in the XY-plane defined by the bolt group coordinate system. The given positions are in relation to the transformation plane in which the bolt group was selected.
            {
                positions.Add(new {
                    x = Math.Round(pos.X, 2),
                    y = Math.Round(pos.Y, 2),
                    z = Math.Round(pos.Z, 2)
                });
            }
            boltData["positions"] = positions;
            boltData["bolt_size"] = boltGroup.BoltSize;
            boltData["bolt_standard"] = boltGroup.BoltStandard;
        }
    }
}



```

```csharp
var relatedObjects = new List<string>();
var relatedEnum = drawingObject.GetRelatedObjects();
while (relatedEnum.MoveNext())
{
    var relatedObj = relatedEnum.Current;
    if (relatedObj != null)
    {
        string typeName = relatedObj.GetType().Name;
        string identifier = "";
        
        // Try to get ModelIdentifier
        var idProp = relatedObj.GetType().GetProperty("ModelIdentifier");
        if (idProp != null)
        {
            var idValue = idProp.GetValue(relatedObj, null);
            if (idValue != null)
            {
                identifier = ":" + idValue.ToString();
            }
        }
        
        relatedObjects.Add(typeName + identifier);
    }
}
```

```csharp
private void PreviewObjectExtraction(DrawingObject drawingObject)
{
    var debugInfo = new StringBuilder();
    var functionsUsed = new List<string>();
    var extractedData = new Dictionary<string, object>();
    
    try
    {
        // Step 1: Basic object info
        txtObjectType.Text = drawingObject.GetType().FullName;
        txtObjectTypeName.Text = drawingObject.GetType().Name;
        
        functionsUsed.Add("✓ GetType() → " + drawingObject.GetType().Name);
        
        // Step 2: Try to get ModelIdentifier
        var modelIdProp = drawingObject.GetType().GetProperty("ModelIdentifier");
        if (modelIdProp != null)
        {
            var modelId = modelIdProp.GetValue(drawingObject, null);
            txtModelId.Text = modelId?.ToString() ?? "N/A";
            functionsUsed.Add("✓ ModelIdentifier → " + modelId);
        }
        
        // Step 3: Route to appropriate extractor
        if (drawingObject is Part part)
        {
            functionsUsed.Add("✓ PartExtractor.ExtractPart()");
            extractedData = ExtractPartWithDebug(part, functionsUsed);
        }
        else if (drawingObject is StraightDimension dimension)
        {
            functionsUsed.Add("✓ DimensionExtractor.ExtractDimension()");
            extractedData = ExtractDimensionWithDebug(dimension, functionsUsed);
        }
        else if (drawingObject is StraightDimensionSet dimensionSet)
        {
            functionsUsed.Add("✓ DimensionExtractor.ExtractDimensionSet()");
            extractedData = ExtractDimensionSetWithDebug(dimensionSet, functionsUsed);
        }
        else if (drawingObject is Bolt bolt)
        {
            functionsUsed.Add("✓ BoltExtractor.ExtractBolt()");
            extractedData = ExtractBoltWithDebug(bolt, functionsUsed);
        }
        else
        {
            functionsUsed.Add("⚠ Generic extraction (no specialized extractor)");
            extractedData = ExtractGenericObjectWithDebug(drawingObject, functionsUsed);
        }
        
        // Step 4: Find parent view
        functionsUsed.Add("✓ ViewExtractor.FindParentView()");
        var parentView = FindParentView(drawingObject);
        if (parentView != null)
        {
            functionsUsed.Add("  └─ Found: " + parentView);
            extractedData["parent_view"] = parentView;
        }
        
        // Step 5: Get related objects
        functionsUsed.Add("✓ RelatedObjectsExtractor.GetRelatedObjects()");
        var relatedObjects = ExtractRelatedObjectsWithDebug(drawingObject, functionsUsed);
        if (relatedObjects.Count > 0)
        {
            functionsUsed.Add($"  └─ Found {relatedObjects.Count} related");
            extractedData["related_objects"] = relatedObjects;
        }
        
        // Display results
        DisplayFunctionsUsed(functionsUsed);
        DisplayExtractedJson(extractedData);
    }
    catch (Exception ex)
    {
        functionsUsed.Add("❌ ERROR: " + ex.Message);
        DisplayFunctionsUsed(functionsUsed);
        ShowError($"Extraction preview failed: {ex.Message}\n\n{ex.StackTrace}");
    }
}
```

```csharp
private Dictionary<string, object> ExtractPartWithDebug(
    Part drawingPart, 
    List<string> debugLog)
{
    var data = new Dictionary<string, object>();
    
    try
    {
        // Get model object
        debugLog.Add("  └─ GetModelObject()");
        var modelObj = new Model().SelectModelObject(drawingPart.ModelObject);
        
        if (modelObj is Beam beam)
        {
            debugLog.Add("    └─ Type: Beam");
            
            // Extract profile
            debugLog.Add("  └─ ExtractProfile()");
            var profile = beam.Profile.ProfileString;
            debugLog.Add($"    └─ Result: \"{profile}\"");
            
            // Extract material
            debugLog.Add("  └─ ExtractMaterial()");
            var material = beam.Material.MaterialString;
            debugLog.Add($"    └─ Result: \"{material}\"");
            
            // Calculate length
            debugLog.Add("  └─ CalculateBeamLength()");
            var length = Math.Round(Distance.PointToPoint(beam.StartPoint, beam.EndPoint), 2);
            debugLog.Add($"    └─ Result: {length} mm");
            
            // Extract geometry
            debugLog.Add("  └─ ExtractGeometry()");
            debugLog.Add("    └─ start_point, end_point, bbox");
            
            // Extract chamfer
            debugLog.Add("  └─ ExtractChamfer()");
            if (beam.StartPointOffset.Dx != 0)
            {
                debugLog.Add($"    └─ Result: {beam.StartPointOffset.Dx} mm (square)");
                data["chamfer"] = beam.StartPointOffset.Dx;
            }
            else
            {
                debugLog.Add("    └─ No chamfer");
            }
            
            // Build data structure
            data["model_object"] = new {
                id = drawingPart.ModelObject.ID,
                profile = profile,
                material = material,
                class_value = beam.Class,
                length = length
            };
            
            data["geometry"] = new {
                part_type_model = "Beam",
                start_point = new { 
                    x = beam.StartPoint.X, 
                    y = beam.StartPoint.Y, 
                    z = beam.StartPoint.Z 
                },
                end_point = new { 
                    x = beam.EndPoint.X, 
                    y = beam.EndPoint.Y, 
                    z = beam.EndPoint.Z 
                }
            };
        }
        else if (modelObj is ContourPlate plate)
        {
            debugLog.Add("    └─ Type: ContourPlate");
            // Similar extraction for plates...
        }
    }
    catch (Exception ex)
    {
        debugLog.Add($"  └─ ❌ ERROR: {ex.Message}");
        data["extraction_error"] = ex.Message;
    }
    
    return data;
}

private Dictionary<string, object> ExtractDimensionWithDebug(
    StraightDimension dimension, 
    List<string> debugLog)
{
    var data = new Dictionary<string, object>();
    
    try
    {
        // Extract coordinates
        debugLog.Add("  └─ ExtractCoordinates()");
        var startPoint = dimension.StartPoint;
        var endPoint = dimension.EndPoint;
        debugLog.Add($"    └─ Start: ({startPoint.X:F1}, {startPoint.Y:F1})");
        debugLog.Add($"    └─ End: ({endPoint.X:F1}, {endPoint.Y:F1})");
        
        // Extract value
        debugLog.Add("  └─ ExtractDimensionValue()");
        var dimensionValue = dimension.Value;
        if (dimensionValue != null)
        {
            var textElement = dimensionValue as Text;
            if (textElement != null)
            {
                debugLog.Add($"    └─ Text: \"{textElement.TextString}\"");
                data["value_text"] = textElement.TextString;
            }
        }
        
        // Calculate value from coordinates
        debugLog.Add("  └─ CalculateDistance()");
        var calculatedValue = Math.Sqrt(
            Math.Pow(endPoint.X - startPoint.X, 2) + 
            Math.Pow(endPoint.Y - startPoint.Y, 2)
        );
        debugLog.Add($"    └─ Calculated: {calculatedValue:F2} mm");
        data["value_calculated"] = Math.Round(calculatedValue, 2);
        
        // Calculate direction
        debugLog.Add("  └─ DetermineDirection()");
        var dx = Math.Abs(endPoint.X - startPoint.X);
        var dy = Math.Abs(endPoint.Y - startPoint.Y);
        string direction = dx > dy ? "horizontal" : "vertical";
        debugLog.Add($"    └─ Direction: {direction}");
        
        data["start_point"] = new { x = startPoint.X, y = startPoint.Y };
        data["end_point"] = new { x = endPoint.X, y = endPoint.Y };
        data["direction"] = direction;
        data["distance_to_line"] = dimension.Distance;
    }
    catch (Exception ex)
    {
        debugLog.Add($"  └─ ❌ ERROR: {ex.Message}");
        data["extraction_error"] = ex.Message;
    }
    
    return data;
}

private Dictionary<string, object> ExtractBoltWithDebug(
    Bolt drawingBolt, 
    List<string> debugLog)
{
    var data = new Dictionary<string, object>();
    
    try
    {
        debugLog.Add("  └─ GetModelObject()");
        var boltGroup = new Model().SelectModelObject(drawingBolt.ModelObject) 
            as BoltGroup;
        
        if (boltGroup != null)
        {
            debugLog.Add("    └─ Type: BoltGroup");
            
            // Extract positions
            debugLog.Add("  └─ ExtractBoltPositions()");
            var positions = new List<object>();
            foreach (Point pos in boltGroup.BoltPositions)
            {
                positions.Add(new { 
                    x = Math.Round(pos.X, 2), 
                    y = Math.Round(pos.Y, 2), 
                    z = Math.Round(pos.Z, 2) 
                });
            }
            debugLog.Add($"    └─ Found: {positions.Count} bolt positions");
            
            // Extract properties
            debugLog.Add("  └─ ExtractBoltProperties()");
            debugLog.Add($"    └─ Size: {boltGroup.BoltSize}");
            debugLog.Add($"    └─ Standard: {boltGroup.BoltStandard}");
            
            data["positions"] = positions;
            data["bolt_size"] = boltGroup.BoltSize;
            data["bolt_standard"] = boltGroup.BoltStandard;
            data["position_count"] = positions.Count;
        }
    }
    catch (Exception ex)
    {
        debugLog.Add($"  └─ ❌ ERROR: {ex.Message}");
        data["extraction_error"] = ex.Message;
    }
    
    return data;
}
```

```csharp
private void DisplayFunctionsUsed(List<string> functions)
{
    txtFunctionsUsed.Clear();
    foreach (var func in functions)
    {
        txtFunctionsUsed.AppendText(func + Environment.NewLine);
    }
    
    // Highlight errors in red
    int index = txtFunctionsUsed.Text.IndexOf("❌");
    while (index >= 0)
    {
        txtFunctionsUsed.Select(index, txtFunctionsUsed.Text.IndexOf('\n', index) - index);
        txtFunctionsUsed.SelectionColor = Color.Red;
        index = txtFunctionsUsed.Text.IndexOf("❌", index + 1);
    }
}

private void DisplayExtractedJson(Dictionary<string, object> data)
{
    var settings = new JsonSerializerSettings
    {
        Formatting = Formatting.Indented,
        ReferenceLoopHandling = ReferenceLoopHandling.Ignore,
        NullValueHandling = NullValueHandling.Ignore
    };
    
    string json = JsonConvert.SerializeObject(data, settings);
    txtExtractedJson.Text = json;
    
    // Syntax highlighting (optional but nice)
    ApplyJsonSyntaxHighlighting(txtExtractedJson);
}

private void ApplyJsonSyntaxHighlighting(RichTextBox rtb)
{
    // Simple syntax highlighting for JSON
    string[] keywords = { "\"type\"", "\"profile\"", "\"material\"", "\"length\"" };
    foreach (var keyword in keywords)
    {
        int index = rtb.Text.IndexOf(keyword);
        while (index >= 0)
        {
            rtb.Select(index, keyword.Length);
            rtb.SelectionColor = Color.Blue;
            index = rtb.Text.IndexOf(keyword, index + keyword.Length);
        }
    }
}
```
