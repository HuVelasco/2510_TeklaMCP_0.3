```python
# New MCP tool for connections
@mcp.tool()
async def create_connection(
    primary_object_id: int,
    secondary_object_ids: List[int], 
    connection_number: int = 144  # Default to end plate
) -> dict:
    """Create connection between structural elements using IDs"""
    payload = {
        "PrimaryObjectId": primary_object_id,
        "SecondaryObjectIds": secondary_object_ids,
        "ConnectionNumber": connection_number
    }
    return await call_endpoint("/create-connection-by-ids", payload=payload)
```
